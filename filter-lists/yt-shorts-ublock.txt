! ============================================================
!  SECTION 2: SEARCH RESULTS — THE CRITICAL FIX
!  These selectors target shorts appearing as regular video
!  cards in search results with a "SHORTS" badge overlay.
! ============================================================

! --- PRIMARY: Hide video cards with SHORTS overlay badge ---
! This is THE key selector. YouTube renders shorts in search
! as ytd-video-renderer with an overlay-style="SHORTS" badge.
www.youtube.com##ytd-thumbnail-overlay-time-status-renderer[overlay-style="SHORTS"]:upward(ytd-video-renderer)

! --- ALTERNATIVE: Direct :has() matching for search results ---
www.youtube.com##ytd-video-renderer:has([overlay-style="SHORTS"])
www.youtube.com##ytd-video-renderer:has(ytd-thumbnail-overlay-time-status-renderer[overlay-style="SHORTS"])

! --- Grid layout search results ---
www.youtube.com##ytd-grid-video-renderer:has([overlay-style="SHORTS"])
www.youtube.com##ytd-grid-video-renderer:has(ytd-thumbnail-overlay-time-status-renderer[overlay-style="SHORTS"])

! --- Rich item renderer (new YouTube layout) ---
www.youtube.com##ytd-rich-item-renderer:has([overlay-style="SHORTS"])
www.youtube.com##ytd-rich-item-renderer:has(ytd-thumbnail-overlay-time-status-renderer[overlay-style="SHORTS"])

! --- New lockup view model (2025-2026 YouTube redesign) ---
www.youtube.com##ytm-shorts-lockup-view-model-v2:upward(grid-shelf-view-model)
www.youtube.com##ytd-rich-item-renderer[rendered-from-rich-grid] ytm-shorts-lockup-view-model:upward(ytd-rich-item-renderer[rendered-from-rich-grid])

! --- Hide videos with #shorts / #short in title ---
www.youtube.com##ytd-video-renderer:has(#video-title:has-text(/(^| )#Shorts?( |$)/i))
www.youtube.com##ytd-grid-video-renderer:has(#video-title:has-text(/(^| )#Shorts?( |$)/i))
www.youtube.com##ytd-rich-item-renderer:has(#video-title:has-text(/(^| )#Shorts?( |$)/i))
www.youtube.com##ytd-compact-video-renderer:has(#video-title:has-text(/(^| )#Shorts?( |$)/i))

! --- Hide shorts by href pattern (catch-all) ---
www.youtube.com##ytd-video-renderer:has(a[href*="/shorts/"])
www.youtube.com##ytd-grid-video-renderer:has(a[href*="/shorts/"])
www.youtube.com##ytd-rich-item-renderer:has(a[href*="/shorts/"])
www.youtube.com##ytd-compact-video-renderer:has(a[href*="/shorts/"])

! --- Hide the Shorts shelf/section in search results ---
www.youtube.com##ytd-search-renderer ytd-reel-shelf-renderer
www.youtube.com##ytd-item-section-renderer:has(ytd-reel-shelf-renderer)
www.youtube.com##ytd-reel-shelf-renderer
www.youtube.com##ytd-reel-item-renderer

! --- Hide Shorts filter chip on search page ---
www.youtube.com##yt-chip-cloud-chip-renderer:has(.ytChipShapeInactive:has-text(/^Shorts$/i))
www.youtube.com##yt-chip-cloud-chip-renderer:has(yt-formatted-string:has-text(/^Shorts$/i))

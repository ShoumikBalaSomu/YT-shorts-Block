! ==========================================================
! 1. STRICT URL & API BLOCKING (Prevents Direct Bypasses)
! ==========================================================
! Block direct navigation to Shorts URLs at the network level
||youtube.com/shorts/$all
||www.youtube.com/shorts/$all
||m.youtube.com/shorts/$all

! Block Shorts-specific API endpoints to stop background loading & recommendations
||youtube.com/youtubei/v1/reel_item_endpoint$xhr
||youtube.com/youtubei/v1/shorts/*$xhr
||youtube.com/youtubei/v1/reel_watch_sequence$xhr

! ==========================================================
! 2. PRUNING SHORTS FROM API RESPONSES (Prevents them in feeds/searches)
! ==========================================================
! Remove Shorts shelves and items from YouTube's initial JSON data
www.youtube.com##+js(json-prune, contents.twoColumnSearchResultsRenderer.primaryContents.sectionListRenderer.contents.*.itemSectionRenderer.contents.*.reelShelfRenderer)
www.youtube.com##+js(json-prune, contents.twoColumnSearchResultsRenderer.primaryContents.sectionListRenderer.contents.*.itemSectionRenderer.contents.*.reelItemRenderer)
www.youtube.com##+js(json-prune, contents.twoColumnBrowseResultsRenderer.tabs.*.tabRenderer.content.richGridRenderer.contents.*.richSectionRenderer)
www.youtube.com##+js(json-prune, contents.twoColumnBrowseResultsRenderer.tabs.*.tabRenderer.content.richGridRenderer.contents.*.richItemRenderer.content.reelItemRenderer)

! Remove Shorts from continuation (infinite scroll) API responses
www.youtube.com##+js(json-prune-fetch-response, onResponseReceivedActions.*.appendContinuationItemsAction.continuationItems.*.reelItemRenderer)
www.youtube.com##+js(json-prune-fetch-response, onResponseReceivedActions.*.appendContinuationItemsAction.continuationItems.*.reelShelfRenderer)

! ==========================================================
! 3. UI ELEMENT HIDING (DESKTOP)
! ==========================================================
! Hide Shorts tab in left sidebar navigation
youtube.com##ytd-guide-entry-renderer[title="Shorts"]
youtube.com##ytd-mini-guide-entry-renderer[title="Shorts"]
youtube.com##ytd-guide-entry-renderer:has-text(/^Shorts$/i)

! Hide Shorts shelves on homepage, search, and subscription feeds
youtube.com##ytd-rich-shelf-renderer[is-shorts]
youtube.com##ytd-reel-shelf-renderer
youtube.com##ytd-rich-section-renderer:has(#title:has-text(/Shorts/i))

! Hide Shorts videos in search results and general feeds (Thumbnails & Titles)
youtube.com##ytd-video-renderer:has([overlay-style="SHORTS"])
youtube.com##ytd-video-renderer:has(span[aria-label="Shorts"])
youtube.com##ytd-video-renderer:has-text(/#shorts/i)
youtube.com##ytd-rich-item-renderer:has(ytd-thumbnail-overlay-time-status-renderer[overlay-style="SHORTS"])
youtube.com##ytd-rich-item-renderer:has-text(/#shorts/i)
youtube.com##ytd-grid-video-renderer:has([overlay-style="SHORTS"])
youtube.com##ytd-grid-video-renderer:has(span[aria-label="ShortS"])
youtube.com##ytd-compact-video-renderer[is-short]

! Hide Shorts tab on channel pages
youtube.com##tp-yt-paper-tab:has-text(/^Shorts$/)
youtube.com##yt-tab-shape:has-text(/^Shorts$/)

! Hide Shorts duration badge on thumbnails
youtube.com##ytd-thumbnail-overlay-time-status-renderer[overlay-style="SHORTS"]
youtube.com##span[aria-label="Shorts"]

! 4. UI ELEMENT HIDING (MOBILE)
! ==========================================================

! 5. ANTI-BYPASS MEASURES
! ==========================================================
! Force block if URL somehow changes to /shorts/ via SPA navigation
youtube.com##+js(aost, history.pushState, /shorts/)
youtube.com##+js(aost, history.replaceState, /shorts/)

! Block attempts to load shorts via embedded or alternative URL parameters
||youtube.com/embed/*?feature=shorts
||youtube.com/watch?*feature=shorts$document

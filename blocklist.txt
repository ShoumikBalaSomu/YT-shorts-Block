! Description: The most advanced filter list to completely block YouTube Shorts, preventing bypasses, hiding all UI elements, and pruning search results.

! ==========================================================
! 1. NETWORK & URL BLOCKING (Blocks direct navigation & API)
! ==========================================================
||youtube.com/shorts/$all
||m.youtube.com/shorts/$all
||www.youtube.com/shorts/$all
||youtube.com/youtubei/v1/reel/*
||youtube.com/youtubei/v1/next*$shorts*
||youtube.com/youtubei/v1/player*$shorts*

! ==========================================================
! 2. SEARCH RESULTS & HOMEPAGE PRUNING (JSON Pruning)
! ==========================================================
! Removes Shorts from search results and homepage shelves at the JSON level
youtube.com##+js(json-prune, response.contents.twoColumnSearchResultsRenderer.primaryContents.sectionListRenderer.contents.*.itemSectionRenderer.contents.*.reelShelfRenderer)
youtube.com##+js(json-prune, response.contents.twoColumnBrowseResultsRenderer.tabs.*.tabRenderer.content.richGridRenderer.contents.*.richSectionRenderer.content.richShelfRenderer)
youtube.com##+js(json-prune, response.contents.twoColumnBrowseResultsRenderer.tabs.*.tabRenderer.content.richGridRenderer.contents.*.richItemRenderer.content.reelItemRenderer)
youtube.com##+js(json-prune-fetch-response, response.contents.twoColumnSearchResultsRenderer.primaryContents.sectionListRenderer.contents.*.itemSectionRenderer.contents.*.reelShelfRenderer)

! ==========================================================
! 3. UI ELEMENT HIDING (DESKTOP)
! ==========================================================
! Hide Shorts tab in left sidebar navigation (Homepage Button)
youtube.com##ytd-guide-entry-renderer:has(> tp-yt-paper-item > yt-formatted-string:has-text(/Shorts/))
youtube.com##ytd-mini-guide-entry-renderer:has(> a > yt-formatted-string:has-text(/Shorts/))
youtube.com##ytd-guide-entry-renderer[title="Shorts"]
youtube.com##ytd-mini-guide-entry-renderer[title="Shorts"]

! Hide Shorts shelf on homepage and subscription feeds
youtube.com##ytd-rich-shelf-renderer[is-shorts]
youtube.com##ytd-rich-section-renderer:has(ytd-rich-shelf-renderer[is-shorts])
youtube.com##ytd-reel-shelf-renderer
youtube.com##ytd-rich-item-renderer:has(ytd-thumbnail-overlay-time-status-renderer[overlay-style="SHORTS"])

! Hide Shorts videos in search results and general feeds
youtube.com##ytd-video-renderer[is-short]
youtube.com##ytd-compact-video-renderer[is-short]
youtube.com##ytd-reel-item-renderer

! Hide Shorts tab on channel pages
youtube.com##tp-yt-paper-tab:has-text(/^Shorts$/)

! Hide Shorts duration badge on thumbnails
youtube.com##ytd-thumbnail-overlay-time-status-renderer[overlay-style="SHORTS"]

! ==========================================================
! 4. UI ELEMENT HIDING (MOBILE)
! ==========================================================
m.youtube.com##ytm-guide-entry-renderer[title="Shorts"]
m.youtube.com##ytm-rich-shelf-renderer[is-shorts]
m.youtube.com##ytm-video-with-context-renderer:has(.ytm-thumbnail-overlay-time-status-renderer[overlay-style="SHORTS"])
m.youtube.com##ytm-tab-header-renderer:has-text(Shorts)
m.youtube.com##ytm-pivot-bar-item-renderer[title="Shorts"]
m.youtube.com##ytm-shorts-lockup-view-model-v2:upward(grid-shelf-view-model)

! ==========================================================
! 5. ANTI-BYPASS SCRIPTLETS
! ==========================================================
! Prevents YouTube's internal SPA router from navigating to /shorts/
youtube.com##+js(trusted-replace-fetch-response, '{"playabilityStatus":{"status":"ERROR"}}', undefined, /\/youtubei\/v1\/player/)
